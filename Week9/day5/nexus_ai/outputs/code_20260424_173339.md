def quick_sort(arr):
    """
    Sorts an array using the Quick Sort algorithm.

    Args:
        arr (list): The array to be sorted.

    Returns:
        list: The sorted array.
    """

    # Base case: If the array has one or zero elements, it is already sorted
    if len(arr) <= 1:
        return arr
    
    # Select the pivot element (in this case, the median of three elements)
    # Alternative pivot selection techniques: first element, last element, random pivot
    pivot_index = ((len(arr) - 1) + (len(arr) // 2)) // 2

    # Move the pivot to the end of the array
    arr.insert(0, arr.pop(pivot_index))  # More efficient than swap

    # Partition the array into three subarrays: elements less than, equal to, and greater than the pivot
    less_than_pivot = [x for i, x in enumerate(arr) if i != 0 and x < arr[0]]
    equal_to_pivot = [x for x in arr if x == arr[0]]
    greater_than_pivot = [x for i, x in enumerate(arr) if i != 0 and x > arr[0]]

    # Recursively sort the subarrays and combine them
    # Use recursion instead of explicit calls to avoid stack overflow errors
    return quick_sort(less_than_pivot) + equal_to_pivot + quick_sort(greater_than_pivot)
